###############################################################################
# 1. NEW NODE - BASIC HOSTNAME / DNS / NETWORK CHECK
###############################################################################

# [NEW NODE]

hostnamectl set-hostname tvsnvrapp024mp.mak.iss

getent hosts tvsnvrapp024mp.mak.iss

ip -br addr

###############################################################################
# 2. NEW NODE - INSTALL REQUIRED PCS/PACEMAKER PACKAGES
###############################################################################

# [NEW NODE]

rpm -q pcs pacemaker corosync resource-agents fence-agents-ipmilan ipmitool

# If packages are missing, install them from your local repository:
#
# dnf install -y pcs pacemaker corosync resource-agents fence-agents-ipmilan ipmitool


###############################################################################
# 3. NEW NODE - START AND ENABLE PCS DAEMON
###############################################################################

# [NEW NODE]

systemctl enable --now pcsd

systemctl status pcsd --no-pager

ss -lntp | grep 2224


###############################################################################
# 4. NEW NODE - ALLOW PCS TCP/2224 THROUGH FIREWALL
###############################################################################

# [NEW NODE]

firewall-cmd --permanent --add-port=2224/tcp
firewall-cmd --reload

firewall-cmd --query-port=2224/tcp

# Expected:
# yes


###############################################################################
# 5. EXISTING NODE - VERIFY CONNECTIVITY TO NEW NODE
###############################################################################

# [EXISTING NODE]

getent hosts tvsnvrapp024mp.mak.iss

nc -zv tvsnvrapp024mp.mak.iss 2224

curl -k -v https://tvsnvrapp024mp.mak.iss:2224/remote/status


###############################################################################
# 6. EXISTING NODE - AUTHENTICATE NEW NODE WITH PCS
###############################################################################

# [EXISTING NODE]

pcs host auth tvsnvrapp024mp.mak.iss -u hacluster

# Enter the hacluster password when prompted.

# Expected:
# tvsnvrapp024mp.mak.iss: Authorized
